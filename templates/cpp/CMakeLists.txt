cmake_minimum_required(VERSION 3.25)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PROJECT_FULL_VERSION LIMIT_COUNT 1)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" PROJECT_NUMERIC_VERSION "${PROJECT_FULL_VERSION}")
if(NOT PROJECT_NUMERIC_VERSION)
  message(FATAL_ERROR "VERSION must start with MAJOR.MINOR.PATCH, found \"${PROJECT_FULL_VERSION}\"")
endif()

option(PROJECT_GUI "Build the dashboard window" ON)
option(PROJECT_TESTS "Build the tests" ON)
option(PROJECT_FUZZ "Build the fuzz targets with libFuzzer (Clang only) instead of as replay programs" OFF)
option(PROJECT_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
option(PROJECT_CHECKED "Keep assertions and checked containers in an optimized build" OFF)
set(PROJECT_SANITIZE "" CACHE STRING "Sanitizers for the build: address, thread or empty")

if(PROJECT_GUI)
  list(APPEND VCPKG_MANIFEST_FEATURES "gui")
endif()
if(PROJECT_TESTS)
  list(APPEND VCPKG_MANIFEST_FEATURES "tests")
endif()
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND DEFINED ENV{VCPKG_ROOT})
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "vcpkg toolchain")
endif()
if(CMAKE_HOST_WIN32 AND NOT DEFINED VCPKG_TARGET_TRIPLET)
  if("$ENV{PROCESSOR_ARCHITECTURE}" STREQUAL "ARM64")
    set(VCPKG_TARGET_TRIPLET "arm64-windows-static" CACHE STRING "vcpkg triplet")
  else()
    set(VCPKG_TARGET_TRIPLET "x64-windows-static" CACHE STRING "vcpkg triplet")
  endif()
endif()

project(templateapp VERSION ${PROJECT_NUMERIC_VERSION} LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/bin>")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(APP_NAME "Template App")
set(APP_SLUG "templateapp")
set(APP_ID "io.github.unalivin.templateapp")
set(APP_PUBLISHER "Unalivin")
set(APP_SUMMARY "TEMPLATE_TAGLINE")
set(APP_REPOSITORY_URL "https://github.com/unalivin/workspace")

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" copyright_lines REGEX "^Copyright \\(c\\) ")
if(NOT copyright_lines)
  message(FATAL_ERROR "LICENSE has no copyright line")
endif()
list(GET copyright_lines 0 APP_COPYRIGHT)
if(NOT APP_COPYRIGHT MATCHES " ${APP_PUBLISHER}$")
  message(FATAL_ERROR "the copyright line of LICENSE does not name the publisher ${APP_PUBLISHER}: ${APP_COPYRIGHT}")
endif()

include(cmake/Embed.cmake)
include(cmake/Quality.cmake)

find_package(nlohmann_json CONFIG REQUIRED)
find_package(ftxui CONFIG REQUIRED)

add_subdirectory(third_party/sqlite)
add_subdirectory(src/core)
add_subdirectory(src/tui)
if(PROJECT_GUI)
  add_subdirectory(src/gui)
endif()
add_subdirectory(src/app)
add_subdirectory(tools)

if(PROJECT_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
add_subdirectory(fuzz)

include(cmake/Package.cmake)
