file(GLOB fuzz_target_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/targets/*.cpp")
list(SORT fuzz_target_sources)

add_library(fuzz_support STATIC support/fuzzing.cpp)
target_include_directories(fuzz_support PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(fuzz_support PUBLIC app_core)
project_quality(fuzz_support)

if(PROJECT_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
    message(FATAL_ERROR "PROJECT_FUZZ needs Clang with libFuzzer, found ${CMAKE_CXX_COMPILER_ID}")
  endif()
else()
  add_library(fuzz_replay OBJECT support/replay_main.cpp)
  target_include_directories(fuzz_replay PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/src")
  project_quality(fuzz_replay)
endif()

set(fuzz_executables "")
foreach(source IN LISTS fuzz_target_sources)
  get_filename_component(name "${source}" NAME_WE)
  set(executable "fuzz_${name}")
  add_executable(${executable} "${source}")
  target_link_libraries(${executable} PRIVATE fuzz_support)
  project_quality(${executable})
  if(PROJECT_FUZZ)
    target_link_options(${executable} PRIVATE -fsanitize=fuzzer)
  else()
    target_sources(${executable} PRIVATE $<TARGET_OBJECTS:fuzz_replay>)
    if(PROJECT_TESTS)
      add_test(NAME ${executable} COMMAND ${executable} "${CMAKE_CURRENT_SOURCE_DIR}/seeds/${name}")
    endif()
  endif()
  list(APPEND fuzz_executables ${executable})
endforeach()
add_custom_target(fuzzers DEPENDS ${fuzz_executables})
